theorem cached_codes (keys : List Key) : (cached keys).map Prod.fst = keys.map code := by
  simp [cached, List.map_map, Function.comp_def]

def missingFamilyCodes : List Nat := missingCodes ++ input.cachedFamily.map Prod.fst
def outsideRequestCodes : List Nat := outsideCodes ++ input.cachedRequest.map Prod.fst

theorem missingFamilyCodes_eq : missingFamilyCodes = (missing ++ familyKeys).map code := by
  rw [missingFamilyCodes, missingCodes_eq, cache_family, cached_codes, List.map_append]

theorem outsideRequestCodes_eq : outsideRequestCodes = (outside ++ requested).map code := by
  rw [outsideRequestCodes, outsideCodes_eq, cache_request, cached_codes, List.map_append]

theorem missing_disjoint_checked :
    FamilyKeyOrder.checkOrder (sortBy (· ≤ ·) missingFamilyCodes) = true := by decide

theorem missing_family_nodup : (missing ++ familyKeys).Nodup := by
  apply FamilyKeyOrder.nodup_of_map_nodup code
  rw [← missingFamilyCodes_eq]
  exact (sortBy_perm (· ≤ ·) missingFamilyCodes).nodup
    (FamilyKeyOrder.checkOrder_nodup _ missing_disjoint_checked)

  exact (sortBy_perm (· ≤ ·) outsideRequestCodes).nodup
    (FamilyKeyOrder.checkOrder_nodup _ outside_disjoint_checked)

theorem requested_nodup : requested.Nodup := (List.nodup_append.mp outside_requested_nodup).2.1
